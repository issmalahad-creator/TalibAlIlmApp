const out = String.fromEnvironment('OUT', defaultValue: '/tmp/preview');

Future<void> _font(String family, List<String> files) async {
  final l = FontLoader(family);
  for (final f in files) {
    l.addFont(Future.value(ByteData.sublistView(File(f).readAsBytesSync())));
  }
  await l.load();
}

Future<void> _shot(WidgetTester t, String name) async {
  await t.runAsync(() async {
    final boundary = find.byType(RepaintBoundary).evaluate().first.renderObject as RenderRepaintBoundary;
    final img = await boundary.toImage(pixelRatio: 1.5);
    final bytes = await img.toByteData(format: ui.ImageByteFormat.png);
    Directory(out).createSync(recursive: true);
    File('$out/$name.png').writeAsBytesSync(bytes!.buffer.asUint8List());
  });
}

Future<void> _settle(WidgetTester t, [int n = 8]) async {
  for (var i = 0; i < n; i++) {
    await t.runAsync(() => Future<void>.delayed(const Duration(milliseconds: 250)));
    await t.pump(const Duration(milliseconds: 150));
  }
}

        home: home,

    );

void main() {
  TestWidgetsFlutterBinding.ensureInitialized();
  sqfliteFfiInit();
  databaseFactory = databaseFactoryFfi;
  final log = StringBuffer();

  setUpAll(() async {
    final m = TestDefaultBinaryMessengerBinding.instance.defaultBinaryMessenger;
    for (final ch in ['xyz.luan/audioplayers.global', 'xyz.luan/audioplayers', 'plugin.csdcorp.com/speech_to_text', 'dexterous.com/flutter/local_notifications', 'flutter.baseflow.com/permissions/methods', 'flutter.baseflow.com/geolocator', 'hemanthraj/flutter_compass', 'dev.fluttercommunity.plus/connectivity', 'wakelock_plus', 'flutter_timezone']) {
      m.setMockMethodCallHandler(MethodChannel(ch), (_) async => null);
    }
    DatabaseHelper.databaseName = 'talib_tour.db';
    final dbFile = File(p.join('.dart_tool', 'sqflite_common_ffi', 'databases', DatabaseHelper.databaseName));
    if (dbFile.existsSync()) dbFile.deleteSync();
    final gz = File(p.join('assets', 'mushaf', 'mushaf_layout.json.gz'));
    await MushafLayoutSync().ingestFromJson(jsonDecode(utf8.decode(gzip.decode(gz.readAsBytesSync()))) as Map<String, dynamic>);
    try { await QuranImportService().importIfNeeded(includeLegacyTafsir: false); } catch (e) { log.writeln('SEED quranImport: $e'); }
    try { await MilestoneRepository().seedIfNeeded(); } catch (e) { log.writeln('SEED milestones: $e'); }
    await _font('Amiri', ['assets/fonts/Amiri-Regular.ttf', 'assets/fonts/Amiri-Bold.ttf']);
    // The UI font is 'Tahoma' (not bundled — the phone falls back to its
    // system Arabic font). Without a real font every letter is a wide box and
    // the overflow check reports false alarms. Download Noto Sans Arabic first:
    // raw.githubusercontent.com/notofonts/notofonts.github.io/main/fonts/NotoSansArabic/hinted/ttf/NotoSansArabic-{Regular,Bold}.ttf
    const noto = String.fromEnvironment('NOTO', defaultValue: '/tmp/fonts');
    await _font('Tahoma', ['$noto/NotoSansArabic-Regular.ttf', '$noto/NotoSansArabic-Bold.ttf']);
    await _font('AmiriQuran', ['assets/fonts/AmiriQuran-Regular.ttf']);
    await _font('MaterialIcons', ['/root/sdk/flutter/bin/cache/artifacts/material_fonts/MaterialIcons-Regular.otf']);
  });

  tearDownAll(() => File('$out/_log.txt').writeAsStringSync(log.toString()));

  for (final (tag, size, scale) in const [('n', Size(400, 860), 1.0), ('s', Size(360, 740), 1.3)]) {
  for (final (name, build) in screens) {
    testWidgets('$tag $name', (t) async {
      SharedPreferences.setMockInitialValues({});
      t.view.physicalSize = size;
      t.view.devicePixelRatio = 1;
      final errors = <String>[];
      final prev = FlutterError.onError;
      FlutterError.onError = (d) {
        final m = d.exceptionAsString().split(String.fromCharCode(10)).first;
        if (!m.contains('MissingPluginException')) errors.add(m);
      };
      try {
        await t.pumpWidget(_app(build(), scale));
        await _settle(t, 12);
        await _shot(t, '${tag}_$name');
      } catch (e) {
        errors.add('THROW: $e'.split(String.fromCharCode(10)).first);
      }
      FlutterError.onError = prev;
      final ex = t.takeException();
      if (ex != null && !'$ex'.contains('MissingPluginException')) errors.add('EXC: $ex'.split(String.fromCharCode(10)).first);
      log.writeln('== $tag $name: ${errors.isEmpty ? 'ok' : errors.toSet().join(' | ')}');
      await t.pumpWidget(const SizedBox());
      await _settle(t, 2);
    }, timeout: const Timeout(Duration(minutes: 2)));
  }
  }
}
